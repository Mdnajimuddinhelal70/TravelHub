export const uploadBufferToCloudinary = async (
  buffer: Buffer,
  fileName: string,
): Promise<UploadApiResponse | undefined> => {
  try {
    return new Promise((resolve, reject) => {
      const public_id = `pdf/${fileName}-${Date.now()}`;

      const bufferStream = new stream.PassThrough();
      bufferStream.end(buffer);

      cloudinary.uploader
        .upload_stream(
          {
            resource_type: "auto",
            public_id: public_id,
            folder: "pdf",
          },
          (error, result) => {
            if (error) {
              return reject(error);
            }
            resolve(result);
          },
        )
        .end(buffer);
    });
  } catch (error: any) {
    console.log(error);
    throw new AppError(401, `Error uploading file ${error.message}`);
  }
};

import multer from "multer";
import { CloudinaryStorage } from "multer-storage-cloudinary-v2";
import { cloudinaryUpload } from "./cloudinary.config";

const storage = new CloudinaryStorage({
  // The adapter's TypeScript definition is incompatible with
  // Cloudinary SDK v2's uploader callback types.
  cloudinary: cloudinaryUpload as unknown as ConstructorParameters<
    typeof CloudinaryStorage
  >[0]["cloudinary"],

  params: {
    public_id: (_req, file) => {
      // Example: My Special.Image#!@.png
      // Result: unique-id-timestamp-my-special-image-png

      const originalName = file.originalname.toLowerCase();

      const fileName = originalName
        .replace(/\.[^.]+$/, "")
        .replace(/\s+/g, "-")
        .replace(/[^a-z0-9-]/g, "")
        .replace(/-+/g, "-")
        .replace(/^-|-$/g, "");

      const extension = file.originalname.split(".").pop()?.toLowerCase();

      const uniqueId = Math.random().toString(36).substring(2);

      const uniqueFileName = [uniqueId, Date.now(), fileName || "upload"].join(
        "-",
      );

      return extension ? `${uniqueFileName}.${extension}` : uniqueFileName;
    },
  },
});

export const multerUpload = multer({
  storage,
});

/* eslint-disable @typescript-eslint/no-explicit-any */
import bcryptjs from "bcryptjs";
import passport from "passport";
import {
  Strategy as GoogleStrategy,
  Profile,
  VerifyCallback,
} from "passport-google-oauth20";
import { Strategy as LocalStrategy } from "passport-local";
import { IsActive, Role } from "../modules/user/user.interface";
import { User } from "../modules/user/user.model";
import { envVars } from "./env";

passport.use(
  new LocalStrategy(
    {
      usernameField: "email",
      passwordField: "password",
    },
    async (email: string, password: string, done) => {
      try {
        const isUserExist = await User.findOne({ email }).select("+password");

        if (!isUserExist) {
          return done(null, false, { message: "User does not exist" });
        }

        if (
          isUserExist.isActive === IsActive.BLOCKED ||
          isUserExist.isActive === IsActive.INACTIVE
        ) {
          return done(null, false, {
            message: `User is ${isUserExist.isActive}`,
          });
        }

        if (isUserExist.isDeleted) {
          return done(null, false, { message: "User is deleted" });
        }

        const isGoogleAuthenticated = isUserExist.auths.some(
          (providerObjects) => providerObjects.provider == "google",
        );

        if (isGoogleAuthenticated && !isUserExist.password) {
          return done(null, false, {
            message:
              "You have authenticated through Google. So if you want to login with credentials, then at first login with google and set a password for your Gmail and then you can login with email and password.",
          });
        }

        const isPasswordMatched = await bcryptjs.compare(
          password as string,
          isUserExist.password as string,
        );

        if (!isPasswordMatched) {
          return done(null, false, { message: "Password does not match" });
        }

        return done(null, isUserExist);
      } catch (error) {
        console.log(error);
        done(error);
      }
    },
  ),
);

passport.use(
  new GoogleStrategy(
    {
      clientID: envVars.GOOGLE_CLIENT_ID,
      clientSecret: envVars.GOOGLE_CLIENT_SECRET,
      callbackURL: envVars.GOOGLE_CALLBACK_URL,
    },
    async (
      accessToken: string,
      refreshToken: string,
      profile: Profile,
      done: VerifyCallback,
    ) => {
      try {
        const email = profile.emails?.[0].value;

        if (!email) {
          return done(null, false, { message: "No email found" });
        }

        let isUserExist = await User.findOne({ email });
        if (
          isUserExist &&
          (isUserExist.isActive === IsActive.BLOCKED ||
            isUserExist.isActive === IsActive.INACTIVE)
        ) {
          return done(null, false, {
            message: `User is ${isUserExist.isActive}`,
          });
        }

        if (isUserExist && isUserExist.isDeleted) {
          return done(null, false, { message: "User is deleted" });
        }

        if (!isUserExist) {
          isUserExist = await User.create({
            email,
            name: profile.displayName,
            picture: profile.photos?.[0].value,
            role: Role.USER,
            auths: [
              {
                provider: "google",
                providerId: profile.id,
              },
            ],
          });
        }

        return done(null, isUserExist);
      } catch (error) {
        console.log("Google Strategy Error", error);
        return done(error);
      }
    },
  ),
);

passport.serializeUser((user: any, done: (err: any, id?: unknown) => void) => {
  done(null, user._id);
});

passport.deserializeUser(async (id: string, done: any) => {
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (error) {
    console.log(error);
    done(error);
  }
});

  err: mongoose.Error.ValidationError,
): TGenericErrorResponse => {
  const errorSources: TErrorSources[] = [];

  const errors = Object.values(err.errors);

  errors.forEach((errorObject: any) =>

      path: errorObject.path,
      message: errorObject.message,

    statusCode: 400,
    message: "Validation Error",
    errorSources,
  };

import { NextFunction, Request, Response } from "express";
import httpStatus from "http-status-codes";
import { JwtPayload } from "jsonwebtoken";
import { envVars } from "../config/env";
import AppError from "../errorHelpers/AppError";
import { IsActive, Role } from "../modules/user/user.interface";
import { User } from "../modules/user/user.model";
import { verifyToken } from "../utils/jwt";

const isValidRole = (role: unknown): role is Role => {
  return Object.values(Role).some((validRole) => validRole === role);
};

export const checkAuth =
  (...authRoles: Role[]) =>
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const authorization = req.headers.authorization;

      const accessToken = authorization?.replace(/^Bearer\s+/i, "").trim();

      if (!accessToken) {
        throw new AppError(
          httpStatus.UNAUTHORIZED,
          "Access token is required.",
        );
      }

      let decoded: string | JwtPayload;

      try {
        decoded = verifyToken(accessToken, envVars.JWT_ACCESS_SECRET) as
          | string
          | JwtPayload;
      } catch {
        throw new AppError(
          httpStatus.UNAUTHORIZED,
          "Invalid or expired access token.",
        );
      }

      if (
        typeof decoded !== "object" ||
        decoded === null ||
        typeof decoded.email !== "string" ||
        !isValidRole(decoded.role)
      ) {
        throw new AppError(
          httpStatus.UNAUTHORIZED,
          "Invalid access token payload.",
        );
      }

      const user = await User.findOne({ email: decoded.email });

      if (!user) {
        throw new AppError(httpStatus.UNAUTHORIZED, "User does not exist.");
      }

      if (user.isDeleted) {
        throw new AppError(httpStatus.FORBIDDEN, "This account is deleted.");
      }

      if (
        user.isActive === IsActive.BLOCKED ||
        user.isActive === IsActive.INACTIVE
      ) {
        throw new AppError(httpStatus.FORBIDDEN, `User is ${user.isActive}.`);
      }

      // Always authorize using the current database role.
      if (!authRoles.includes(user.role)) {
        throw new AppError(
          httpStatus.FORBIDDEN,
          "You are not permitted to access this resource.",
        );
      }

      req.user = {
        ...decoded,
        userId: user._id.toString(),
        email: user.email,
        role: user.role,
      } as JwtPayload;

      next();
    } catch (error) {
      next(error);
    }
  };

  let message = "Something Went Wrong!!";

  //Duplicate error
  if (err.code === 11000) {
    const simplifiedError = handlerDuplicateError(err);
    statusCode = simplifiedError.statusCode;
    message = simplifiedError.message;
  }
  // Object ID error / Cast Error
  else if (err.name === "CastError") {
    const simplifiedError = handleCastError(err);
    statusCode = simplifiedError.statusCode;
    message = simplifiedError.message;
  } else if (err.name === "ZodError") {
    const simplifiedError = handlerZodError(err);
    statusCode = simplifiedError.statusCode;
    message = simplifiedError.message;
    errorSources = simplifiedError.errorSources as TErrorSources[];
  }
  //Mongoose Validation Error
  else if (err.name === "ValidationError") {
    const simplifiedError = handlerValidationError(err);
    statusCode = simplifiedError.statusCode;
    errorSources = simplifiedError.errorSources as TErrorSources[];
    message = simplifiedError.message;
  } else if (err instanceof AppError) {
    statusCode = err.statusCode;
    message = err.message;
  } else if (err instanceof Error) {
    statusCode = 500;
    message = err.message;
  }

  res.status(statusCode).json({
    success: false,
    message,
    errorSources,
    err: envVars.NODE_ENV === "development" ? err : null,
    stack: envVars.NODE_ENV === "development" ? err.stack : null,
  });
};

import bcryptjs from "bcryptjs";
import httpStatus from "http-status-codes";
import jwt, { JwtPayload } from "jsonwebtoken";
import { envVars } from "../../config/env";
import AppError from "../../errorHelpers/AppError";
import { sendEmail } from "../../utils/sendEmail";
import {
  createNewAccessTokenWithRefreshToken,
  createUserTokens,
} from "../../utils/userTokens";
import { IAuthProvider, IsActive, IUser } from "../user/user.interface";
import { User } from "../user/user.model";

const credentialsLogin = async (payload: Partial<IUser>) => {
  const { email, password } = payload;

  const isUserExist = await User.findOne({ email });

  if (!isUserExist) {
    throw new AppError(httpStatus.BAD_REQUEST, "Email does not exist");
  }

  const isPasswordMatched = await bcryptjs.compare(
    password as string,
    isUserExist.password as string,
  );

  if (!isPasswordMatched) {
    throw new AppError(httpStatus.BAD_REQUEST, "Incorrect Password");
  }

  const userTokens = createUserTokens(isUserExist);
  const { password: pass, ...rest } = isUserExist.toObject();

  return {
    accessToken: userTokens.accessToken,
    refreshToken: userTokens.refreshToken,
    user: rest,
  };
};

const getNewAccessToken = async (refreshToken: string) => {
  const newAccessToken =
    await createNewAccessTokenWithRefreshToken(refreshToken);

  return {
    accessToken: newAccessToken,
  };
};
const resetPassword = async (
  payload: Record<string, any>,
  decodedToken: JwtPayload,
) => {
  if (payload.id != decodedToken.userId) {
    throw new AppError(401, "You can not reset your password");
  }

  const isUserExist = await User.findById(decodedToken.userId);
  if (!isUserExist) {
    throw new AppError(401, "User does not exist");
  }

  const hashedPassword = await bcryptjs.hash(
    payload.newPassword,
    Number(envVars.BCRYPT_SALT_ROUND),
  );

  isUserExist.password = hashedPassword;

  await isUserExist.save();
};
const forgotPassword = async (email: string) => {
  const isUserExist = await User.findOne({ email });

  if (!isUserExist) {
    throw new AppError(httpStatus.BAD_REQUEST, "User does not exist");
  }
  if (
    isUserExist.isActive === IsActive.BLOCKED ||
    isUserExist.isActive === IsActive.INACTIVE
  ) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      `User is ${isUserExist.isActive}`,
    );
  }
  if (isUserExist.isDeleted) {
    throw new AppError(httpStatus.BAD_REQUEST, "User is deleted");
  }

  const jwtPayload = {
    userId: isUserExist._id,
    email: isUserExist.email,
    role: isUserExist.role,
  };

  const resetToken = jwt.sign(jwtPayload, envVars.JWT_ACCESS_SECRET, {
    expiresIn: "10m",
  });

  const resetUILink = `${envVars.FRONTEND_URL}/reset-password?id=${isUserExist._id}&token=${resetToken}`;

  sendEmail({
    to: isUserExist.email,
    subject: "Password Reset",
    templateName: "forgetPassword",
    templateData: {
      name: isUserExist.name,
      resetUILink,
    },
  });
};
const setPassword = async (userId: string, plainPassword: string) => {
  const user = await User.findById(userId);

  if (!user) {
    throw new AppError(404, "User not found");
  }

  if (
    user.password &&
    user.auths.some((providerObject) => providerObject.provider === "google")
  ) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      "You have already set you password. Now you can change the password from your profile password update",
    );
  }

  const hashedPassword = await bcryptjs.hash(
    plainPassword,
    Number(envVars.BCRYPT_SALT_ROUND),
  );

  const credentialProvider: IAuthProvider = {
    provider: "credentials",
    providerId: user.email,
  };

  const auths: IAuthProvider[] = [...user.auths, credentialProvider];

  user.password = hashedPassword;

  user.auths = auths;

  await user.save();
};
const changePassword = async (
  oldPassword: string,
  newPassword: string,
  decodedToken: JwtPayload,
) => {
  const user = await User.findById(decodedToken.userId);

  const isOldPasswordMatch = await bcryptjs.compare(
    oldPassword,
    user!.password as string,
  );
  if (!isOldPasswordMatch) {
    throw new AppError(httpStatus.UNAUTHORIZED, "Old Password does not match");
  }

  user!.password = await bcryptjs.hash(
    newPassword,
    Number(envVars.BCRYPT_SALT_ROUND),
  );

  user!.save();
};

import httpStatus from "http-status-codes";
import mongoose from "mongoose";
import AppError from "../../errorHelpers/AppError";
import { getTransactionId } from "../../utils/getTransactionId";

import { PAYMENT_STATUS } from "../payment/payment.interface";
import { Payment } from "../payment/payment.model";
import { ISSLCommerz } from "../sslCommerz/sslCommerz.interface";
import { SSLService } from "../sslCommerz/sslCommerz.service";
import {
  TOUR_DEPARTURE_STATUS as DEPARTURE_STATUS,
  TOUR_DEPARTURE_STATUS,
} from "../tour-departure/tour-departure.interface";
import { TourDeparture } from "../tour-departure/tour-departure.model";
import { User } from "../user/user.model";
import { BOOKING_STATUS, IBooking } from "./booking.interface";
import { Booking } from "./booking.model";

const RESERVATION_DURATION_MS = 15 * 60 * 1000;

const createBooking = async (payload: Partial<IBooking>, userId: string) => {
  const guestCount = Number(payload.guestCount);
  const departureId = payload.tourDeparture;

  if (!departureId || !mongoose.isValidObjectId(departureId)) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      "A valid tour departure is required.",
    );
  }

  if (!Number.isInteger(guestCount) || guestCount < 1) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      "Guest count must be at least 1.",
    );
  }

  const user = await User.findById(userId);

  if (!user) {
    throw new AppError(httpStatus.NOT_FOUND, "User not found.");
  }

  if (!user.phone || !user.address) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      "Please update your profile before booking a tour.",
    );
  }

  const session = await mongoose.startSession();

  let bookingId: mongoose.Types.ObjectId;
  let transactionId: string;
  let amount: number;

  try {
    session.startTransaction();

    // Atomically reserve seats to prevent overbooking.
    const departure = await TourDeparture.findOneAndUpdate(
      {
        _id: departureId,
        status: DEPARTURE_STATUS.OPEN,
        availableSeats: { $gte: guestCount },
      },
      {
        $inc: { availableSeats: -guestCount },
      },
      {
        new: true,
        runValidators: true,
        session,
      },
    );

    if (!departure) {
      throw new AppError(
        httpStatus.CONFLICT,
        "Departure is unavailable or there are not enough seats.",
      );
    }

    if (departure.availableSeats === 0) {
      departure.status = TOUR_DEPARTURE_STATUS.FULL;
      await departure.save({ session });
    }

    const pricePerPerson = departure.pricePerPerson;
    amount = pricePerPerson * guestCount;

    if (!Number.isFinite(amount) || amount <= 0) {
      throw new AppError(httpStatus.BAD_REQUEST, "Invalid booking amount.");
    }

    const reservationExpiresAt = new Date(Date.now() + RESERVATION_DURATION_MS);

    const [booking] = await Booking.create(
      [
        {
          user: user._id,
          tourDeparture: departure._id,
          guestCount,
          pricePerPerson,
          totalAmount: amount,
          status: BOOKING_STATUS.PENDING,
          reservationExpiresAt,
        },
      ],
      { session },
    );

    transactionId = getTransactionId();

    const [payment] = await Payment.create(
      [
        {
          booking: booking._id,
          transactionId,
          amount,
          currency: "BDT",
          status: PAYMENT_STATUS.PENDING,
        },
      ],
      { session },
    );

    await Booking.findByIdAndUpdate(
      booking._id,
      { payment: payment._id },
      {
        new: true,
        runValidators: true,
        session,
      },
    );

    bookingId = booking._id;

    await session.commitTransaction();
  } catch (error) {
    if (session.inTransaction()) {
      await session.abortTransaction();
    }

    throw error;
  } finally {
    await session.endSession();
  }

  // Call the external gateway only after the database transaction commits.
  const sslPayload: ISSLCommerz = {
    address: user.address,
    email: user.email,
    phoneNumber: user.phone,
    name: user.name,
    amount,
    transactionId: transactionId!,
  };

  const sslPayment = await SSLService.sslPaymentInit(sslPayload);

  const booking = await Booking.findById(bookingId!)
    .populate("user", "name email phone address")
    .populate({
      path: "tourDeparture",
      populate: {
        path: "tour",
        select: "title slug",
      },
    })
    .populate("payment");

  return {
    paymentUrl: sslPayment.GatewayPageURL,
    booking,
  };
};

  gender: z.enum(["male", "female", "other"]),
  type: z.enum(["adult", "child"]),
  nidOrPassport: z.string().optional(),

  name: z.string().min(1, "Contact name is required").trim(),
  phone: z.string().min(1, "Phone is required").trim(),
  email: z.string().email("Invalid email").trim().toLowerCase(),
  emergencyContact: z.string().optional(),
});

 *  BOOKING — CREATE
 * ============================================================ */

export const createBookingZodSchema = z
  .object({
    tourDeparture: z.string().min(1, "Tour departure is required"),

    travelers: z
      .array(travelerSchema)
      .min(1, "At least one traveler is required"),

    adultCount: z.number().int().min(1, "At least one adult is required"),
    childCount: z.number().int().min(0).optional(),

    contactInfo: contactInfoSchema,

    couponCode: z.string().trim().toUpperCase().optional(),
    specialRequests: z.string().optional(),

    // ⚠️ বাদ: bookingNumber, user, tour, payment, guestCount,
    //        priceBreakdown, status, reservationExpiresAt
    //        — এগুলো সব server তৈরি করবে
  })
  .refine(
    (data) => {
      const expected = data.adultCount + (data.childCount ?? 0);
      return data.travelers.length === expected;
    },
    {
      message: "travelers count must equal adultCount + childCount",
      path: ["travelers"],
    },
  )
  .refine(
    (data) => {
      const adultTravelers = data.travelers.filter(
        (t) => t.type === "adult",
      ).length;
      return adultTravelers === data.adultCount;
    },
    {
      message: "adultCount must match travelers with type 'adult'",
      path: ["adultCount"],
    },
  );

/* ============================================================
 *  BOOKING — UPDATE STATUS (admin)
 * ============================================================ */

import { deleteImageFromCLoudinary } from "../../config/cloudinary.config";
import { QueryBuilder } from "../../utils/QueryBuilder";
import { divisionSearchableFields } from "./division.constant";
import { IDivision } from "./division.interface";
import { Division } from "./division.model";

/* ============================================================
 *  HELPER — slug generator
 * ============================================================ */
const generateSlug = (name: string): string => {
  return name
    .toLowerCase()
    .trim()
    .replace(/[^\w\u0980-\u09FF\s-]/g, "") // Bengali + English + digits
    .replace(/[\s_-]+/g, "-")
    .replace(/^-+|-+$/g, "");
};

/* ============================================================
 *  CREATE
 * ============================================================ */
const createDivision = async (payload: IDivision) => {
  const existingDivision = await Division.findOne({ name: payload.name });
  if (existingDivision) {
    throw new Error("A division with this name already exists.");
  }

  // slug না দিলে auto-generate
  if (!payload.slug) {
    payload.slug = generateSlug(payload.name);
  }

  // slug uniqueness check
  const existingSlug = await Division.findOne({ slug: payload.slug });
  if (existingSlug) {
    throw new Error("A division with this slug already exists.");
  }

  const division = await Division.create(payload);
  return division;
};

/* ============================================================
 *  GET ALL (paginated)
 * ============================================================ */
const getAllDivisions = async (query: Record<string, string>) => {
  const queryBuilder = new QueryBuilder(Division.find(), query);

  const divisionsData = queryBuilder
    .search(divisionSearchableFields)
    .filter()
    .sort()
    .fields()
    .paginate();

  const [data, meta] = await Promise.all([
    divisionsData.build(),
    queryBuilder.getMeta(),
  ]);

  return { data, meta };
};

/* ============================================================
 *  GET SINGLE
 * ============================================================ */
const getSingleDivision = async (slug: string) => {
  const division = await Division.findOne({ slug });
  if (!division) {
    throw new Error("Division not found.");
  }
  return { data: division };
};

/* ============================================================
 *  UPDATE
 * ============================================================ */
const updateDivision = async (id: string, payload: Partial<IDivision>) => {
  const existingDivision = await Division.findById(id);
  if (!existingDivision) {
    throw new Error("Division not found.");
  }

  // ✅ FIX: name থাকলে এবং সত্যিই বদলালে তবেই duplicate check
  if (payload.name && payload.name !== existingDivision.name) {
    const duplicateDivision = await Division.findOne({
      name: payload.name,
      _id: { $ne: id },
    });
    if (duplicateDivision) {
      throw new Error("A division with this name already exists.");
    }

    // name বদলালে slug-ও regenerate (user explicitly slug না দিলে)
    if (!payload.slug) {
      payload.slug = generateSlug(payload.name);
    }
  }

  // ✅ slug explicitly দিলে uniqueness check
  if (payload.slug && payload.slug !== existingDivision.slug) {
    const duplicateSlug = await Division.findOne({
      slug: payload.slug,
      _id: { $ne: id },
    });
    if (duplicateSlug) {
      throw new Error("A division with this slug already exists.");
    }
  }

  const updatedDivision = await Division.findByIdAndUpdate(id, payload, {
    new: true,
    runValidators: true,
  });

  // নতুন thumbnail এলে পুরনোটা Cloudinary থেকে মুছুন
  if (payload.thumbnail && existingDivision.thumbnail) {
    await deleteImageFromCLoudinary(existingDivision.thumbnail);
  }

  return updatedDivision;
};

/* ============================================================
 *  DELETE
 * ============================================================ */
const deleteDivision = async (id: string) => {
  const division = await Division.findById(id);
  if (!division) {
    throw new Error("Division not found.");
  }

  // TODO: referential integrity — Destination/Tour এই division ব্যবহার করছে কি না চেক করুন
  // const hasDestinations = await Destination.exists({ division: id });
  // if (hasDestinations) {
  //   throw new Error("Cannot delete division with existing destinations.");
  // }

  await Division.findByIdAndDelete(id);

  // ✅ Cloudinary থেকে thumbnail মুছুন
  if (division.thumbnail) {
    await deleteImageFromCLoudinary(division.thumbnail);
  }

  return null;
};

      type: Date,

import AppError from "../../errorHelpers/AppError";

import { generatePdf, IInvoiceData } from "../../utils/invoice";

const initPayment = async (bookingId: string) => {
  const payment = await Payment.findOne({ booking: bookingId });

  if (!payment) {
    throw new AppError(
      httpStatus.NOT_FOUND,
      "Payment Not Found. You have not booked this tour",
    );
  }

  const booking = await Booking.findById(payment.booking);

  const userAddress = (booking?.user as any).address;
  const userEmail = (booking?.user as any).email;
  const userPhoneNumber = (booking?.user as any).phone;
  const userName = (booking?.user as any).name;

  const sslPayload: ISSLCommerz = {
    address: userAddress,
    email: userEmail,
    phoneNumber: userPhoneNumber,
    name: userName,
    amount: payment.amount,
    transactionId: payment.transactionId,
  };

  const sslPayment = await SSLService.sslPaymentInit(sslPayload);

  return {
    paymentUrl: sslPayment.GatewayPageURL,
  };
};
const successPayment = async (query: Record<string, string>) => {
  // Update Booking Status to COnfirm
  // Update Payment Status to PAID

  const session = await Booking.startSession();
  session.startTransaction();

  try {
    const updatedPayment = await Payment.findOneAndUpdate(
      { transactionId: query.transactionId },
      {
        status: PAYMENT_STATUS.PAID,
      },
      { new: true, runValidators: true, session: session },
    );

    if (!updatedPayment) {
      throw new AppError(401, "Payment not found");
    }

    const updatedBooking = await Booking.findByIdAndUpdate(
      updatedPayment?.booking,
      { status: BOOKING_STATUS.COMPLETED },
      { new: true, runValidators: true, session },
    )
      .populate({
        path: "tourDeparture",
        populate: {
          path: "tour",
          select: "title slug",
        },
      })
      .populate("user", "name email");

    if (!updatedBooking) {
      throw new AppError(401, "Booking not found");
    }

    const invoiceData: IInvoiceData = {
      bookingDate: updatedBooking.createdAt as Date,
      guestCount: updatedBooking.guestCount,
      totalAmount: updatedPayment.amount,
      tourTitle: (updatedBooking?.tourDeparture as any)?.tour?.title ?? "Tour",
      transactionId: updatedPayment.transactionId,
      userName: (updatedBooking.user as unknown as IUser).name,
    };

    const pdfBuffer = await generatePdf(invoiceData);

    const cloudinaryResult = await uploadBufferToCloudinary(
      pdfBuffer,
      "invoice",
    );

    if (!cloudinaryResult) {
      throw new AppError(401, "Error uploading pdf");
    }

    await Payment.findByIdAndUpdate(
      updatedPayment._id,
      { invoiceUrl: cloudinaryResult.secure_url },
      { runValidators: true, session },
    );

    await sendEmail({
      to: (updatedBooking.user as unknown as IUser).email,
      subject: "Your Booking Invoice",
      templateName: "invoice",
      templateData: invoiceData,
      attachments: [
        {
          filename: "invoice.pdf",
          content: pdfBuffer,
          contentType: "application/pdf",
        },
      ],
    });

    await session.commitTransaction(); //transaction
    session.endSession();
    return { success: true, message: "Payment Completed Successfully" };
  } catch (error) {
    await session.abortTransaction(); // rollback
    session.endSession();
    // throw new AppError(httpStatus.BAD_REQUEST, error) ❌❌
    throw error;
  }
};

type UpdateTourPayload = Partial<ITour> & {
  deleteImages?: string[];

/* ============================================================
 *  HELPER — slug generator
 * ============================================================ */
const generateSlug = (text: string): string => {
  return text
    .toLowerCase()
    .trim()
    .replace(/[^\w\u0980-\u09FF\s-]/g, "")
    .replace(/[\s_-]+/g, "-")
    .replace(/^-+|-+$/g, "");
};

/* ============================================================
 *  TOUR — CREATE
 * ============================================================ */
const createTour = async (payload: ITour) => {
  // slug auto-generate
  if (!payload.slug) {
    payload.slug = generateSlug(payload.title);
  }

  const existingTour = await Tour.findOne({
    $or: [{ title: payload.title }, { slug: payload.slug }],
  });
  if (existingTour) {
    throw new Error("A tour with this title or slug already exists.");
  }

  // ✅ referential integrity — destinations exist করে কি না
  if (payload.destinations?.length) {
    const foundDestinations = await Destination.countDocuments({
      _id: { $in: payload.destinations },
    });
    if (foundDestinations !== payload.destinations.length) {
      throw new Error("One or more destinations not found.");
    }
  }

  // ✅ tourType exist করে কি না
  if (payload.tourType) {
    const foundType = await TourType.findById(payload.tourType);
    if (!foundType) {
      throw new Error("Tour type not found.");
    }
  }

  return await Tour.create(payload);
};

/* ============================================================
 *  TOUR — GET ALL
 * ============================================================ */
const getAllTours = async (query: Record<string, string>) => {
  const queryBuilder = new QueryBuilder(Tour.find(), query);

  const tours = queryBuilder
    .search(tourSearchableFields)
    .filter()
    .sort()
    .fields()
    .paginate();

  const [data, meta] = await Promise.all([
    tours
      .build()
      .populate("destinations", "name slug thumbnail")
      .populate("tourType", "name slug icon"),
    queryBuilder.getMeta(),
  ]);

  return { data, meta };
};

const getSingleTour = async (slug: string) => {
  const tour = await Tour.findOne({ slug })
    .populate("destinations", "name slug thumbnail description")
    .populate("tourType", "name slug icon");

  if (!tour) {
    throw new Error("Tour not found.");
  }

  return tour; // ✅ wrapper বাদ
};

/* ============================================================
 *  TOUR — UPDATE
 * ============================================================ */
const updateTour = async (id: string, payload: UpdateTourPayload) => {
  const existingTour = await Tour.findById(id);
  if (!existingTour) {
    throw new Error("Tour not found.");
  }

  const { deleteImages = [], images: newImages, ...updateData } = payload;

  const finalUpdateData: Partial<ITour> = { ...updateData };

  // ✅ title বদলালে slug regenerate + duplicate check
  if (updateData.title && updateData.title !== existingTour.title) {
    const newSlug = updateData.slug || generateSlug(updateData.title);
    const duplicate = await Tour.findOne({
      _id: { $ne: id },
      $or: [{ title: updateData.title }, { slug: newSlug }],
    });
    if (duplicate) {
      throw new Error("A tour with this title or slug already exists.");
    }
    finalUpdateData.slug = newSlug;
  }

  // ✅ destinations validate
  if (updateData.destinations?.length) {
    const found = await Destination.countDocuments({
      _id: { $in: updateData.destinations },
    });
    if (found !== updateData.destinations.length) {
      throw new Error("One or more destinations not found.");
    }
  }

  // ✅ tourType validate
  if (updateData.tourType) {
    const foundType = await TourType.findById(updateData.tourType);
    if (!foundType) {
      throw new Error("Tour type not found.");
    }
  }

  // ✅ images merge logic
  if (newImages !== undefined || deleteImages.length > 0) {
    const deletedSet = new Set(deleteImages);
    const remaining = (existingTour.images ?? []).filter(
      (url) => !deletedSet.has(url),
    );
    const uploaded = (newImages ?? []).filter((url) => !deletedSet.has(url));
    finalUpdateData.images = [...remaining, ...uploaded];
  }

  const updatedTour = await Tour.findByIdAndUpdate(id, finalUpdateData, {
    new: true,
    runValidators: true,
  });

  // Cloudinary cleanup — update সফল হলে
  if (deleteImages.length > 0) {
    await Promise.all(
      deleteImages.map((url) => deleteImageFromCLoudinary(url)),
    );
  }

  return updatedTour;
};

 *  TOUR — DELETE
 * ============================================================ */
const deleteTour = async (id: string) => {
  const tour = await Tour.findById(id);
  if (!tour) {
    throw new Error("Tour not found.");
  }

  // TODO: referential integrity — কোনো booking থাকলে delete আটকান

  // if (hasBookings) throw new Error("Cannot delete tour with existing bookings.");

  await Tour.findByIdAndDelete(id);

  // ✅ Cloudinary cleanup
  if (tour.images?.length) {
    await Promise.all(tour.images.map((url) => deleteImageFromCLoudinary(url)));
  }

  return null;
};

const updateTourType = async (id: string, payload: Partial<ITourType>) => {
  const existingTourType = await TourType.findById(id);
  if (!existingTourType) {
    throw new Error("Tour type not found.");
  }

  if (Object.keys(payload).length === 0) {
    return existingTourType;
  }

  // ✅ name বদলালে duplicate check + slug regenerate
  if (payload.name && payload.name !== existingTourType.name) {
    const duplicate = await TourType.findOne({
      _id: { $ne: id },
      name: payload.name,
    });
    if (duplicate) {
      throw new Error("A tour type with this name already exists.");
    }
    if (!payload.slug) {
      payload.slug = generateSlug(payload.name);
    }
  }

  return await TourType.findByIdAndUpdate(id, payload, {
    new: true,
    runValidators: true,
  });
};

 *  TOUR TYPE — DELETE

const deleteTourType = async (id: string) => {

    throw new Error("Tour type not found.");
  }

  // ✅ referential integrity — কোনো tour এই type ব্যবহার করছে কি না
  const tourCount = await Tour.countDocuments({ tourType: id });
  if (tourCount > 0) {
    throw new Error(
      `Cannot delete tour type. ${tourCount} tour(s) are using it.`,
    );
  }

  return await TourType.findByIdAndDelete(id);
};

tourDepartureSchema.index({ status: 1, startDate: 1 });
tourDepartureSchema.index({ tour: 1, status: 1 });

// endDate > startDate
tourDepartureSchema.pre("validate", function () {
  if (this.startDate && this.endDate && this.endDate <= this.startDate) {
    throw new Error("End date must be after start date");
  }
});

// availableSeats + bookedSeats = totalSeats
tourDepartureSchema.pre("validate", function () {
  if (this.availableSeats + this.bookedSeats !== this.totalSeats) {
    throw new Error("availableSeats + bookedSeats must equal totalSeats");
  }
});

// availableSeats 0 হলে FULL, ফিরে এলে OPEN
tourDepartureSchema.pre("save", function () {
  if (
    this.isModified("availableSeats") &&
    this.availableSeats === 0 &&
    this.status === TOUR_DEPARTURE_STATUS.OPEN
  ) {
    this.status = TOUR_DEPARTURE_STATUS.FULL;
  }
  if (
    this.isModified("availableSeats") &&
    this.availableSeats > 0 &&
    this.status === TOUR_DEPARTURE_STATUS.FULL
  ) {
    this.status = TOUR_DEPARTURE_STATUS.OPEN;
  }
});

const createTourDeparture = async (payload: ITourDeparture) => {
  // ✅ tour exist করে কি না
  const tour = await Tour.findById(payload.tour);
  if (!tour) {
    throw new Error("Tour not found.");
  }

  // ✅ date validation
  if (new Date(payload.endDate) <= new Date(payload.startDate)) {
    throw new Error("End date must be after start date");
  }

  // ✅ একই tour + startDate-এ duplicate check
  const existing = await TourDeparture.findOne({
    tour: payload.tour,
    startDate: payload.startDate,
  });
  if (existing) {
    throw new Error("A departure already exists for this tour on this date.");
  }

  // ✅ availableSeats না দিলে totalSeats-ই হবে
  if (payload.availableSeats === undefined) {
    payload.availableSeats = payload.totalSeats;
  }
  if (payload.bookedSeats === undefined) {
    payload.bookedSeats = 0;
  }

  // ✅ seat arithmetic validation
  if (payload.availableSeats + payload.bookedSeats !== payload.totalSeats) {
    throw new Error("availableSeats + bookedSeats must equal totalSeats");
  }

  return await TourDeparture.create(payload);
};

 *  UPDATE
 * ============================================================ */
const updateTourDeparture = async (
  id: string,
  payload: Partial<ITourDeparture>,
) => {
  const existing = await TourDeparture.findById(id);
  if (!existing) {
    throw new Error("Tour departure not found.");
  }

  // ✅ date validation (একটা বা দুটোই বদলালে)
  const startDate = payload.startDate ?? existing.startDate;
  const endDate = payload.endDate ?? existing.endDate;
  if (new Date(endDate) <= new Date(startDate)) {
    throw new Error("End date must be after start date");
  }

  // ✅ totalSeats বদলালে availableSeats recalculate
  if (payload.totalSeats !== undefined) {
    if (payload.totalSeats < existing.bookedSeats) {
      throw new Error(
        `Cannot reduce total seats below booked seats (${existing.bookedSeats}).`,
      );
    }
    payload.availableSeats = payload.totalSeats - existing.bookedSeats;
  }

  // ✅ tour বা startDate বদলালে duplicate check
  if (payload.tour || payload.startDate) {
    const duplicate = await TourDeparture.findOne({
      _id: { $ne: id },
      tour: payload.tour ?? existing.tour,
      startDate: payload.startDate ?? existing.startDate,
    });
    if (duplicate) {
      throw new Error("A departure already exists for this tour on this date.");
    }
  }

  // ✅ tour বদলালে exist করে কি না
  if (payload.tour) {
    const tour = await Tour.findById(payload.tour);
    if (!tour) {
      throw new Error("Tour not found.");
    }
  }

  const updated = await TourDeparture.findByIdAndUpdate(id, payload, {
    new: true,
    runValidators: true,
  });

  return updated;
};

/* ============================================================
 *  DELETE
 * ============================================================ */
const deleteTourDeparture = async (id: string) => {
  const departure = await TourDeparture.findById(id);
  if (!departure) {
    throw new Error("Tour departure not found.");
  }

  // ✅ booking থাকলে delete আটকান
  const bookingCount = await Booking.countDocuments({ tourDeparture: id });
  if (bookingCount > 0) {
    throw new Error(
      `Cannot delete departure. ${bookingCount} booking(s) exist.`,
    );
  }

  await TourDeparture.findByIdAndDelete(id);
  return null;
};

/* ============================================================
 *  HELPER — atomic seat decrement (Booking service-এ ব্যবহার হবে)
 * ============================================================ */
const reserveSeats = async (departureId: string, seatCount: number) => {
  const departure = await TourDeparture.findOneAndUpdate(
    {
      _id: departureId,
      status: {
        $in: [TOUR_DEPARTURE_STATUS.OPEN, TOUR_DEPARTURE_STATUS.SCHEDULED],
      },
      availableSeats: { $gte: seatCount },
    },
    {
      $inc: {
        availableSeats: -seatCount,
        bookedSeats: seatCount,
      },
    },
    { new: true },
  );

  if (!departure) {
    throw new Error("Not enough seats available.");
  }

  // ✅ FULL status auto-update (findOneAndUpdate hook চলে না)
  if (departure.availableSeats === 0) {
    await TourDeparture.updateOne(
      { _id: departureId },
      { $set: { status: TOUR_DEPARTURE_STATUS.FULL } },
    );
  }

  return departure;
};

/* ============================================================
 *  HELPER — seat release (booking cancel/expire হলে)
 * ============================================================ */
const releaseSeats = async (departureId: string, seatCount: number) => {
  const departure = await TourDeparture.findByIdAndUpdate(
    departureId,
    {
      $inc: {
        availableSeats: seatCount,
        bookedSeats: -seatCount,
      },
    },
    { new: true },
  );

  if (!departure) {
    throw new Error("Tour departure not found.");
  }

  // ✅ FULL থেকে OPEN-এ ফেরত
  if (
    departure.availableSeats > 0 &&
    departure.status === TOUR_DEPARTURE_STATUS.FULL
  ) {
    await TourDeparture.updateOne(
      { _id: departureId },
      { $set: { status: TOUR_DEPARTURE_STATUS.OPEN } },
    );
  }

  return departure;
};

import bcryptjs from "bcryptjs";
import httpStatus from "http-status-codes";
import { JwtPayload } from "jsonwebtoken";
import { envVars } from "../../config/env";
import AppError from "../../errorHelpers/AppError";

const createUser = async (payload: CreateUserInput) => {
  const email = payload.email.trim().toLowerCase();

  const isUserExist = await User.findOne({ email }).select("+password");

  if (isUserExist) {
    throw new AppError(
      httpStatus.CONFLICT,
      "An account with this email already exists.",
    );
  }

  const saltRounds = Number(envVars.BCRYPT_SALT_ROUND);

  if (!Number.isInteger(saltRounds) || saltRounds < 1) {
    throw new AppError(
      httpStatus.INTERNAL_SERVER_ERROR,
      "Password hashing configuration is invalid.",
    );
  }

  const hashedPassword = await bcryptjs.hash(payload.password, saltRounds);

  const authProvider: IAuthProvider = {
    provider: "credentials",
    providerId: email,
  };

  try {
    const user = await User.create({
      name: payload.name.trim(),
      email,
      password: hashedPassword,
      phone: payload.phone,
      address: payload.address,

      // Public registration must never choose its own role.
      role: Role.USER,

      auths: [authProvider],
    });

    // Explicitly return only safe fields.
    return {
      _id: user._id,
      name: user.name,
      email: user.email,
      phone: user.phone,
      picture: user.picture,
      address: user.address,
      role: user.role,
      isActive: user.isActive,
      createdAt: user.createdAt,
    };
  } catch (error: unknown) {
    // MongoDB duplicate-key error.
    if (
      typeof error === "object" &&
      error !== null &&
      "code" in error &&
      error.code === 11000
    ) {
      throw new AppError(
        httpStatus.CONFLICT,
        "An account with this email already exists.",
      );
    }

    throw error;
  }
};

  const ifUserExist = await User.findById(userId);

  if (!ifUserExist) {
    throw new AppError(httpStatus.NOT_FOUND, "User Not Found");
  }

  if (
    decodedToken.role === Role.ADMIN &&
    ifUserExist.role === Role.SUPER_ADMIN
  ) {
    throw new AppError(401, "You are not authorized");
  }

  if (payload.role) {
    if (decodedToken.role === Role.USER || decodedToken.role === Role.GUIDE) {

      throw new AppError(httpStatus.FORBIDDEN, "You are not authorized");
    }
  }

  const newUpdatedUser = await User.findByIdAndUpdate(userId, payload, {
    new: true,
    runValidators: true,
  });

  return newUpdatedUser;

      const buffer: Uint8Array[] = [];

      doc.text(`Guests: ${invoiceData.guestCount}`);
      doc.text(`Total Amount: $${invoiceData.totalAmount.toFixed(2)}`);
      doc.moveDown();

      doc.text("Thank you for booking with us!", { align: "center" });

      doc.end();

===== ./src/app/utils/QueryBuilder.ts =====
import { Query } from "mongoose";
import { excludeField } from "../constant";

export class QueryBuilder<T> {
  public modelQuery: Query<T[], T>;
  public readonly query: Record<string, string>;

  constructor(modelQuery: Query<T[], T>, query: Record<string, string>) {
    this.modelQuery = modelQuery;
    this.query = query;
  }

  filter(): this {
    const filter = { ...this.query };

    for (const field of excludeField) {
      // eslint-disable-next-line @typescript-eslint/no-dynamic-delete
      delete filter[field];
    }

    this.modelQuery = this.modelQuery.find(filter); // Tour.find().find(filter)

    return this;
  }

  search(searchableField: string[]): this {
    const searchTerm = this.query.searchTerm || "";
    const searchQuery = {
      $or: searchableField.map((field) => ({
        [field]: { $regex: searchTerm, $options: "i" },
      })),
    };
    this.modelQuery = this.modelQuery.find(searchQuery);
    return this;
  }

  sort(): this {
    const sort = this.query.sort || "-createdAt";

    this.modelQuery = this.modelQuery.sort(sort);

    return this;
  }
  fields(): this {
    const fields = this.query.fields?.split(",").join(" ") || "";

    this.modelQuery = this.modelQuery.select(fields);

    return this;
  }
  paginate(): this {
    const page = Number(this.query.page) || 1;
    const limit = Number(this.query.limit) || 10;
    const skip = (page - 1) * limit;

    this.modelQuery = this.modelQuery.skip(skip).limit(limit);

    return this;
  }

  build() {
    return this.modelQuery;
  }

  async getMeta() {
    const totalDocuments = await this.modelQuery.model.countDocuments();

    const page = Number(this.query.page) || 1;
    const limit = Number(this.query.limit) || 10;

    const totalPage = Math.ceil(totalDocuments / limit);

    return { page, limit, total: totalDocuments, totalPage };
  }
}

import httpStatus from "http-status-codes";
import { JwtPayload } from "jsonwebtoken";
import { envVars } from "../config/env";
import AppError from "../errorHelpers/AppError";
import { IsActive, IUser } from "../modules/user/user.interface";
import { User } from "../modules/user/user.model";
import { generateToken, verifyToken } from "./jwt";

export const createUserTokens = (user: Partial<IUser>) => {
  const jwtPayload = {
    userId: user._id,
    email: user.email,
    role: user.role,
  };
  const accessToken = generateToken(
    jwtPayload,
    envVars.JWT_ACCESS_SECRET,
    envVars.JWT_ACCESS_EXPIRES,
  );

  const refreshToken = generateToken(
    jwtPayload,
    envVars.JWT_REFRESH_SECRET,
    envVars.JWT_REFRESH_EXPIRES,
  );

  return {
    accessToken,
    refreshToken,
  };
};

export const createNewAccessTokenWithRefreshToken = async (
  refreshToken: string,
) => {
  const verifiedRefreshToken = verifyToken(
    refreshToken,
    envVars.JWT_REFRESH_SECRET,
  ) as JwtPayload;

  const isUserExist = await User.findOne({ email: verifiedRefreshToken.email });

  if (!isUserExist) {
    throw new AppError(httpStatus.BAD_REQUEST, "User does not exist");
  }
  if (
    isUserExist.isActive === IsActive.BLOCKED ||
    isUserExist.isActive === IsActive.INACTIVE
  ) {
    throw new AppError(
      httpStatus.BAD_REQUEST,
      `User is ${isUserExist.isActive}`,
    );
  }
  if (isUserExist.isDeleted) {
    throw new AppError(httpStatus.BAD_REQUEST, "User is deleted");
  }

  const jwtPayload = {
    userId: isUserExist._id,
    email: isUserExist.email,
    role: isUserExist.role,
  };
  const accessToken = generateToken(
    jwtPayload,
    envVars.JWT_ACCESS_SECRET,
    envVars.JWT_ACCESS_EXPIRES,
  );

  return accessToken;
};
